cmake_minimum_required(VERSION 3.20)
project(intranet_iopv LANGUAGES CXX)
add_library(intranet_iopv STATIC core.cpp)
target_compile_features(intranet_iopv PUBLIC cxx_std_17)
target_include_directories(intranet_iopv PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(intranet_iopv PRIVATE -Wall -Wextra -Werror)
include(CTest)
option(BUILD_LIVE_FEED "Build standalone TGW reader" ON)
if(BUILD_LIVE_FEED)
 find_package(OpenSSL REQUIRED)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(FEED_DEPS REQUIRED IMPORTED_TARGET libzstd simdjson)
 add_library(tgw_reader STATIC third_party/tgw_cpp/src/config.cpp third_party/tgw_cpp/src/protocol.cpp third_party/tgw_cpp/src/session.cpp third_party/tgw_cpp/src/websocket.cpp)
 target_compile_features(tgw_reader PUBLIC cxx_std_20)
 target_include_directories(tgw_reader PUBLIC third_party/tgw_cpp/include PRIVATE third_party/tgw_cpp/src)
 target_link_libraries(tgw_reader PUBLIC OpenSSL::SSL OpenSSL::Crypto PkgConfig::FEED_DEPS)
 add_executable(iopv-feed native/feed.cpp)
 target_link_libraries(iopv-feed PRIVATE tgw_reader)
endif()
if(BUILD_TESTING)
  add_executable(native_core_test tests/native_core_test.cpp)
  target_link_libraries(native_core_test PRIVATE intranet_iopv)
  add_test(NAME native_core_abi COMMAND native_core_test)
endif()
