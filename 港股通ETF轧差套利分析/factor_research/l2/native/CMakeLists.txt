cmake_minimum_required(VERSION 3.19)
project(etf_l2 VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ETF_L2_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(ETF_L2_BUILD_PYTHON "Build pybind11 extension" OFF)
add_library(etf_l2_core STATIC src/engine.cpp src/wind_csv.cpp)
set_target_properties(etf_l2_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(etf_l2_core PUBLIC include)
if(MSVC)
  target_compile_options(etf_l2_core PRIVATE /W4 /WX)
else()
  target_compile_options(etf_l2_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ETF_L2_SANITIZE AND NOT MSVC)
  target_compile_options(etf_l2_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(etf_l2_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(etf_l2_cli src/main.cpp)
target_link_libraries(etf_l2_cli PRIVATE etf_l2_core)
enable_testing()
add_executable(etf_l2_tests tests/test_engine.cpp)
target_link_libraries(etf_l2_tests PRIVATE etf_l2_core)
add_test(NAME engine_contract COMMAND etf_l2_tests)
if(ETF_L2_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND "${Python_EXECUTABLE}" -c "import pybind11; print(pybind11.get_cmake_dir())" OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(etf_l2 python/bindings.cpp)
  target_link_libraries(etf_l2 PRIVATE etf_l2_core)
  add_test(NAME python_binding COMMAND "${Python_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p test_binding.py -v)
  add_test(NAME intraday_batch COMMAND "${Python_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p test_batch.py -v)
  set_tests_properties(python_binding intraday_batch PROPERTIES ENVIRONMENT "PYTHONPATH=$<TARGET_FILE_DIR:etf_l2>")
endif()
